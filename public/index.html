ເຊື່ອມຕໍ່ API ຜົນຫວຍ ແບບ Real-time ຟຣີ!

ເວັບໄຊ້ອື່ນໆ ສາມາດດຶງເອົາຜົນຫວຍຫຼ້າສຸດໄປໃຊ້ງານໄດ້ໃນທັນທີ ໂດຍໃຊ້ REST API ຫຼື ຮັບຂໍ້ມູນ Real-time ຜ່ານ Server-Sent Events (SSE) ເມື່ອມີການກົດແຈ້ງຜົນ.

Endpoints ທີ່ເປີດໃຫ້ບໍລິການ

GET /api/results/latest

ດຶງຜົນຫວຍລາວ ແລະ ໄທ ລ້າສຸດ

GET /api/results/lao

ດຶງປະຫວັດຜົນຫວຍລາວທັງໝົດ

GET /api/results/thai

ດຶງປະຫວັດຜົນຫວຍໄທທັງໝົດ

SSE /api/stream

ເຊື່ອມຕໍ່ຮັບຂໍ້ມູນຜົນຫວຍແບບ Real-time (Server-Sent Events)

ຕົວຢ່າງການຂຽນ Code ດຶງຂໍ້ມູນ Real-time (JavaScript)


// ປ່ຽນ URL ນີ້ໃຫ້ເປັນ Domain ຂອງລະບົບ Lotto Hub ຕົວຈິງຂອງທ່ານ (ເຊັ່ນ: https://lottohub-api.com)
const LOTTO_API_BASE = 'https://web-lottery-realtime.onrender.com';

// 1. ດຶງຂໍ້ມູນຜົນຫວຍລ້າສຸດໃນຄັ້ງທຳອິດ (REST API)
async function fetchLatestLottery() {
  try {
    const response = await fetch(`${LOTTO_API_BASE}/api/results/latest`);
    const data = await response.json();
    console.log('Latest Lottery Results:', data);
  } catch (error) {
    console.error('Error fetching data:', error);
  }
}

// 2. ເຊື່ອມຕໍ່ຮັບຜົນຫວຍ Real-time (Server-Sent Events)
function connectRealtimeStream() {
  const eventSource = new EventSource(`${LOTTO_API_BASE}/api/stream`);

  eventSource.addEventListener('connected', (event) => {
    const data = JSON.parse(event.data);
    console.log('SSE Stream Connected:', data.message);
  });

  eventSource.addEventListener('update', (event) => {
    const update = JSON.parse(event.data);
    console.log('ໄດ້ຮັບຜົນຫວຍໃໝ່ ແບບ Real-time:', update);
    
    // ປະຕິບັດການອັບເດດໜ້າເວັບຂອງທ່ານຢູ່ບ່ອນນີ້
    if (update.type === 'lao') {
      alert(`ຫວຍລາວງວດທີ ${update.data.drawNumber} ອອກ: ${update.data.numbers.join(' ')}`);
    } else if (update.type === 'thai') {
      alert(`ຫວຍໄທ ລາງວັນທີ 1 ອອກ: ${update.data.firstPrize}`);
    }
  });

  eventSource.onerror = (error) => {
    console.error('SSE Error, reconnecting...');
  };
}

// ເລີ່ມເຮັດວຽກ
fetchLatestLottery();
connectRealtimeStream();
